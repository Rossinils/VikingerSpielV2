shader_type canvas_item;
render_mode unshaded;

// Nebelbank der Kampfbuehne (Prototyp): Schwellen auf dem vorhandenen Rauschen (assets/cloud_noise.png),
// senkrecht ausgeblendet, im Bayer-Raster gedithert: harte Pixel statt weicher Wolke, driftet langsam.

uniform sampler2D noise_tex : filter_linear, repeat_enable;
uniform vec4 col : source_color = vec4(0.62, 0.72, 0.74, 0.5);
uniform float drift = 0.0;
uniform float dens : hint_range(0.0, 1.0) = 0.6;
uniform float yscale = 0.03;

varying vec2 lp;

void vertex() {
	lp = VERTEX;
}

const float BAYER[16] = float[16](0.0, 8.0, 2.0, 10.0, 12.0, 4.0, 14.0, 6.0, 3.0, 11.0, 1.0, 9.0, 15.0, 7.0, 13.0, 5.0);

float bayer4(vec2 p) {
	int x = int(mod(p.x, 4.0));
	int y = int(mod(p.y, 4.0));
	return (BAYER[y * 4 + x] + 0.5) / 16.0;
}

void fragment() {
	vec2 p = floor(lp + 0.002);
	vec2 uv = vec2((p.x + drift) * 0.0045, p.y * yscale);
	float n = texture(noise_tex, uv).r * 0.7 + texture(noise_tex, uv * 2.3 + vec2(0.4, 0.1)).g * 0.3;
	float band = 1.0 - abs(UV.y * 2.0 - 1.0);   // Mitte dicht, Raender duenn
	float v = smoothstep(0.35, 0.8, n * (0.55 + band * 0.7)) * dens;
	float q = floor(v * 4.0 + bayer4(p)) / 4.0;
	COLOR = vec4(col.rgb, col.a * clamp(q, 0.0, 1.0));
}
