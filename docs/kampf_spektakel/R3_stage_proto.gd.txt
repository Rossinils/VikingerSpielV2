## PROTOTYP der Kampfbühne (Sandbox "buehne", nicht fürs Hauptprojekt gedacht, nur als Beweis der Technik).
## Sitzt UNTER dem Hud (eigener Platz im Szenenbaum, das Hud liegt in CanvasLayer 1) und besitzt alles, was gezeichnet wird,
## aber nichts, was gedrückt wird (mouse_filter überall IGNORE, Klickflächen bleiben im Hud):
##   Rückwand (Himmel, Meer, Gegnerschiff, Nebel)  ->  Deck mit Requisiten  ->  Figuren (je ein Sprite2D mit Shader)
##   ->  GradeRect (Bildschirm-Shader: Rampenlicht, Abdunkeln, Entsättigen, Farbstich je Ort)  ->  Effekte (FightFx)
##   ->  Glühen (additiv)  ->  Vordergrund (Tau, Fässer, Regen).
## Alle Schichten haben eine Tiefe (0 fern .. 1 Figuren .. 1,45 nah): die Kamera (Zoom um die Bodenlinie, Schwenk in x)
## wirkt je Schicht verschieden stark. Das Hud fragt actor_xf() ab und legt Namen, Balken und Klickflächen durch dieselbe Matrix.
## opt (Dictionary, aus --stage=a,b:c): nograde, noglow, nopar, nocam, nospot, noline, allspot, stepglow, gradeover (Abstufung ÜBER den Figuren, Vorversion), nocell, nofog,
## zoom:1.5, dark:0.5, bd:2 (Maßstab des Gegnerschiffs), theme:kraken, zq:0.125 (Zoomstufe, 0 = stufenlos).

const PAD := 2  # transparenter Rand um jede Figur, damit der Umriss nicht abgeschnitten wird

const DEPTH_BACK := 0.3
const DEPTH_FRONT := 1.45
const BACK_SHIFT := -66.0  # die Weltschichten (Wasserlinie 174) werden so hoch geschoben, dass der Horizont hinter der Reling liegt

class DrawLayer extends Node2D:
	var fn: Callable

	func _draw() -> void:
		if fn.is_valid():
			fn.call(self)


class FighterNode extends Node2D:
	var f: Combat.Fighter
	var spr: Sprite2D
	var mat: ShaderMaterial
	var big := false
	var cur_dim := 0.0
	var cur_desat := 0.0

	func _draw() -> void:
		if big:
			return
		draw_rect(Rect2(-14, -1, 28, 3), Color(0, 0, 0, 0.45))
		draw_rect(Rect2(-10, 2, 20, 1), Color(0, 0, 0, 0.28))

## Standard: die Abstufung liegt UNTER den Figuren (Rückwand und Deck werden dunkel, Figuren dunkeln sich selbst über ihr Material ab).
## Darüber (--stage=gradeover) würde das Dither-Raster auch Blitz und Umriss der Figuren zerstippeln (siehe R3_f_grade_ueber.png).
func _grade_under() -> bool:
	return not opt.has("gradeover")


func _make_fog(r: Rect2, dens: float, drift0: float) -> ColorRect:
	var fog := ColorRect.new()
	fog.mouse_filter = Control.MOUSE_FILTER_IGNORE
	fog.position = r.position
	fog.size = r.size
	var m := ShaderMaterial.new()
	m.shader = load("res://shaders/stage_fog.gdshader")
	m.set_shader_parameter("noise_tex", _noise)
	m.set_shader_parameter("dens", dens)
	m.set_shader_parameter("drift", drift0)
	fog.material = m
	_fog_mats.append(m)
	return fog


## Glühtextur: radialer Abfall; bands > 0 teilt ihn in Stufen (mit Bayer-Dithering im Pixelraster).
func _make_glow(r: int, bands: int, dither: bool) -> Texture2D:
	var img := Image.create(r * 2, r * 2, false, Image.FORMAT_RGBA8)
	const BAYER := [0, 8, 2, 10, 12, 4, 14, 6, 3, 11, 1, 9, 15, 7, 13, 5]
	for y in r * 2:
		for x in r * 2:
			var d := Vector2(float(x) + 0.5 - float(r), float(y) + 0.5 - float(r)).length() / float(r)
			var v := clampf(1.0 - d, 0.0, 1.0)
			v = v * v
			if bands > 0:
				var bm := (float(BAYER[(y % 4) * 4 + (x % 4)]) + 0.5) / 16.0 if dither else 0.5
				v = floorf(v * float(bands) + bm) / float(bands)
			img.set_pixel(x, y, Color(1, 1, 1, clampf(v, 0.0, 1.0)))
	return ImageTexture.create_from_image(img)


func _apply_theme(key: String) -> void:
	if key == th_key:
		return
	th_key = key
	th = THEMES.get(key, THEMES["haendler"])
	_grade_mat.set_shader_parameter("tint", th["tint"])
	_grade_mat.set_shader_parameter("tint_amt", th["tint_amt"])
	_grade_mat.set_shader_parameter("dark", float(opt.get("dark", th["dark"])))
	_grade_mat.set_shader_parameter("desat", th["desat"])
	_grade_mat.set_shader_parameter("warm", th["warm"])
	_grade_mat.set_shader_parameter("vig", 0.55)
	for m in _fog_mats:
		m.set_shader_parameter("col", Color(Color(th["tint"]).lerp(Color(0.8, 0.85, 0.85), 0.55), 0.5))
	_rain.visible = float(th["storm"]) > 0.5


## Vom Hud jedes Bild: der Kampf, wer gerade als Ziel gewählt werden kann.
func sync(c: Combat, valid_foes: Array, valid_heroes: Array) -> void:
	_c = c
	_valid_foes = valid_foes
	_valid_heroes = valid_heroes


## Die Bildverschiebung der Figurenschicht (Tiefe 1): das Hud legt Namen, Balken, Zahlen und Klickflächen durch dieselbe Matrix.
func actor_xf() -> Transform2D:
	return _xf_cache


func _layer_xf(depth: float) -> Transform2D:
	var z := cam_zoom
	var t := _cam_t(z, cam_focus)
	var s := 1.0 + (z - 1.0) * depth
	var o := (t * depth + _shake * depth).round()
	return Transform2D(Vector2(s, 0.0), Vector2(0.0, s), o)


## Zoom um die Bodenlinie (die Füße bleiben auf ihrer Zeile), x: der Brennpunkt wandert Richtung Bildmitte.
func _cam_t(z: float, f: Vector2) -> Vector2:
	var u := clampf(z - 1.0, 0.0, 1.0) * 0.5
	var sx := lerpf(f.x, 240.0, u)
	var tx := sx - f.x * z
	return Vector2(clampf(tx, Waves.VIEW_W * (1.0 - z), 0.0), FEET * (1.0 - z))


func _fx_x(f: Combat.Fighter) -> float:
	return hud._fig_x(f)


func _process(delta: float) -> void:
	_t += delta
	var live := voy != null and voy.phase == Voyage.Phase.COMBAT and voy.combat != null
	visible = live
	if world != null:
		world.visible = not live
	if not live or hud == null:
		return
	var c: Combat = voy.combat
	if c != _c:
		_c = c
		_apply_theme(String(opt.get("theme", c.formation)))
		_valid_foes = []
		_valid_heroes = []
	_collect_valid()
	Waves.storm = float(th["storm"])
	_update_camera(delta)
	_update_layers()
	_update_fighters(delta)
	_update_grade(delta)
	for m in _fog_mats:
		m.set_shader_parameter("drift", _t * 6.0)
	for l in [_l_deck, _l_fx, _l_glow]:
		(l as Node2D).queue_redraw()
	for ch in _l_back.get_children():
		if ch is DrawLayer:
			ch.queue_redraw()
	for ch in _l_front.get_children():
		if ch is DrawLayer:
			ch.queue_redraw()


## Zielwahl wie im Hud (gleiche Regeln), damit die Umrisse stimmen.
func _collect_valid() -> void:
	_valid_foes = []
	_valid_heroes = []
	var c := _c
	if c.state != "hero" or c.actor == null:
		return
	var sk: String = hud._fight_skill
	if sk != "":
		var tl: Array = c.targets(c.actor, sk)
		if String(Roles.skill(sk)["tgt"]) == "ally":
			_valid_heroes = tl
		else:
			_valid_foes = tl
	elif hud._fight_move:
		for hh in c.heroes:
			if hh != c.actor:
				_valid_heroes.append(hh)


func _update_camera(delta: float) -> void:
	# Wer greift an, wer wird getroffen? Daraus Brennpunkt, Zoom und Rampenlicht.
	var attacker: Combat.Fighter = null
	var targets: Array = []
	for f in hud._anim.keys():
		var a: Dictionary = hud._anim[f]
		if float(a.get("wait", 0.0)) > 0.0 and String(a["kind"]) != "attack":
			continue
		match String(a["kind"]):
			"attack":
				attacker = f
			"hit", "dodge":
				targets.append(f)
	_focus_a = null
	_focus_t = []
	var zmax := float(opt.get("zoom", 1.5))
	if attacker != null:
		_focus_a = attacker
		_focus_t = targets
		var ax := _fx_x(attacker)
		var tx := ax
		if targets.size() > 0:
			tx = 0.0
			for tf in targets:
				tx += _fx_x(tf)
			tx /= float(targets.size())
		var span := absf(tx - ax) + 120.0
		goal_zoom = clampf(420.0 / span, 1.0, zmax)
		goal_focus = Vector2((ax + tx) * 0.5, FEET - 34.0)
		spot_goal = Vector2(ax if targets.is_empty() else (ax + tx) * 0.5, FEET - 30.0)
	elif _c.state != "done" and _c.actor != null:
		_focus_a = _c.actor
		var px := _fx_x(_c.actor)
		goal_zoom = 1.0 + (zmax - 1.0) * float(opt.get("idlezoom", 0.0))
		goal_focus = Vector2(px, FEET - 34.0)
		spot_goal = Vector2(px, FEET - 30.0)
	else:
		goal_zoom = 1.0
		goal_focus = Vector2(240.0, 100.0)
		spot_goal = Vector2(240.0, 90.0)
	if opt.has("nocam"):
		goal_zoom = 1.0
	var k := 1.0 - exp(-delta * 8.0)
	cam_zoom = lerpf(cam_zoom, goal_zoom, k)
	cam_focus = cam_focus.lerp(goal_focus, k)
	spot_pos = spot_pos.lerp(spot_goal, k)
	_shake = hud._fx.offset()


func _update_layers() -> void:
	var zq := float(opt.get("zq", 0.125))
	var zshown := cam_zoom * (hud._fx.zoom_scale() if not opt.has("nocam") else 1.0)
	if zq > 0.0:
		zshown = roundf(zshown / zq) * zq
	var saved := cam_zoom
	cam_zoom = zshown
	_xf_cache = _layer_xf(1.0)
	var xf_back := _layer_xf(0.0 if opt.has("nopar") else DEPTH_BACK)
	var xf_front := _layer_xf(1.0 if opt.has("nopar") else DEPTH_FRONT)
	cam_zoom = saved
	_l_back.transform = Transform2D(xf_back.x, xf_back.y, xf_back.origin + Vector2(0.0, BACK_SHIFT * xf_back.y.y).round())
	for n in [_l_deck, _l_act, _l_fx, _l_glow]:
		(n as Node2D).transform = _xf_cache
	_l_front.transform = xf_front


func _update_grade(_delta: float) -> void:
	var sp := _xf_cache * spot_pos
	_grade_mat.set_shader_parameter("spot", sp)
	_grade_mat.set_shader_parameter("spot_r", Vector2(150.0, 78.0) * _xf_cache.x.x)
	_grade_mat.set_shader_parameter("cell", 1.0 if opt.has("nocell") else maxf(_xf_cache.x.x, 1.0))  # Dither-Raster wächst mit dem Zoom mit
	var want := 1.0 if (_focus_a != null and not opt.has("nospot")) else 0.0
	spot_amt = lerpf(spot_amt, want, 0.15)
	_grade_mat.set_shader_parameter("spot_amt", spot_amt if not opt.has("allspot") else 1.0)
	_grade_mat.set_shader_parameter("dark", float(opt.get("dark", th["dark"])) * (1.0 if not opt.has("allspot") else 0.0))


# --- Figuren ---------------------------------------------------------------------------------------------

func _tex(img: Image, key: String, ox: int, oy: int, w: int, h: int) -> Texture2D:
	if _tex_cache.has(key):
		return _tex_cache[key]
	var out := Image.create(w + 2 * PAD, h + 2 * PAD, false, Image.FORMAT_RGBA8)
	out.blit_rect(img, Rect2i(ox, oy, w, h), Vector2i(PAD, PAD))
	var t := ImageTexture.create_from_image(out)
	_tex_cache[key] = t
	return t


func _node_for(f: Combat.Fighter) -> FighterNode:
	if _nodes.has(f):
		return _nodes[f]
	var n := FighterNode.new()
	n.f = f
	n.big = hud._is_big(f)
	n.spr = Sprite2D.new()
	n.spr.centered = false
	n.mat = ShaderMaterial.new()
	n.mat.shader = load("res://shaders/fighter_fx.gdshader")
	n.spr.material = n.mat
	n.add_child(n.spr)
	if n.big:
		var bw := int(Foes.BIG[f.kind]["w"])
		n.spr.offset = Vector2(-bw * 0.5 - PAD, -94.0 - PAD)
		_l_act.add_child(n)
		_l_act.move_child(n, 0)
	else:
		n.spr.offset = Vector2(-FIG_W * 0.5 - PAD, -52.0 - PAD)
		_l_act.add_child(n)
	_nodes[f] = n
	return n


func _update_fighters(delta: float) -> void:
	var alive: Array = []
	for f in _c.heroes:
		alive.append(f)
	for f in _c.foes:
		alive.append(f)
	var ghost_for := {}
	for g in hud._ghosts:
		ghost_for[g["f"]] = g
		if not alive.has(g["f"]):
			alive.append(g["f"])
	for f in _nodes.keys():
		if not alive.has(f):
			(_nodes[f] as Node).queue_free()
			_nodes.erase(f)
	var k := 1.0 - exp(-delta * 10.0)
	var tint_col: Color = th["tint"]
	var tint_amt := float(th["tint_amt"]) * 0.7
	var all_lit := opt.has("allspot") or opt.has("nospot")
	for f in alive:
		var n := _node_for(f)
		var is_ghost: bool = ghost_for.has(f)
		var a: Dictionary = hud._anim.get(f, {})
		if not a.is_empty() and float(a.get("wait", 0.0)) > 0.0:
			a = {}
		var dir := 1.0 if f.side == 0 else -1.0
		var dx := 0.0
		var col: int = (int(Time.get_ticks_msec() / 520) + int(f.rank)) % 2
		if f.side == 0 and f.door:
			col = 4
		var flash := 0.0
		var hit_tint := 0.0
		if not a.is_empty():
			var tt := float(a["t"])
			match String(a["kind"]):
				"attack":
					col = 2
					dx = dir * 9.0 * float(a.get("mv", 1.0)) * sin(PI * clampf(1.0 - tt / 0.42, 0.0, 1.0))
				"dodge":
					dx = -dir * 5.0 * sin(PI * clampf(1.0 - tt / 0.3, 0.0, 1.0))
				_:
					col = 3
					dx = -dir * 2.0 + (1.0 if int(tt * 40.0) % 2 == 0 else -1.0)
					hit_tint = 1.0
					flash = clampf((tt - 0.30) / 0.12, 0.0, 1.0)  # die ersten ~0,12 s ganz weiss
		if n.big and f.charge and a.is_empty():
			col = 4
		var alpha := 1.0
		if is_ghost:
			var g: Dictionary = ghost_for[f]
			var span := 1.2 if n.big else 0.7
			var held := float(g["t"]) > span
			var al := clampf(float(g["t"]) / span, 0.0, 1.0)
			col = 0 if held else 3
			if not held:
				alpha = al
				hit_tint = 1.0
		# Bild wählen
		if n.big:
			var bw := int(Foes.BIG[f.kind]["w"])
			var bh := int(Foes.BIG[f.kind]["h"])
			n.spr.texture = _tex(_kraken_img, "k%d" % col, col * bw, 0, bw, bh)
		else:
			var row := hud._fig_row(f)
			n.spr.texture = _tex(_combat_img, "c%d_%d" % [row, col], col * FIG_W, row * FIG_H, FIG_W, FIG_H)
		n.position = Vector2(_fx_x(f), FEET)
		n.spr.position = Vector2(roundf(dx), 0.0)
		n.mat.set_shader_parameter("alpha", alpha)
		# Wer nicht im Licht steht, wird zusätzlich blasser (nur wenn die Abstufung unter den Figuren liegt)
		var lit: bool = all_lit or f == _focus_a or _focus_t.has(f) or _c.state == "done"
		var dim_goal := 0.0 if (lit or not _grade_under()) else float(opt.get("dim", 0.42))
		var desat_goal := 0.0 if (lit or not _grade_under()) else 0.35
		n.cur_dim = lerpf(n.cur_dim, dim_goal, k)
		n.cur_desat = lerpf(n.cur_desat, desat_goal, k)
		n.mat.set_shader_parameter("dim", n.cur_dim)
		n.mat.set_shader_parameter("desat", n.cur_desat)
		n.mat.set_shader_parameter("flash", flash)
		if hit_tint > 0.0:
			n.mat.set_shader_parameter("tint", Color(1.0, 0.35, 0.32, 0.85))
		else:
			n.mat.set_shader_parameter("tint", Color(tint_col, tint_amt * (0.5 if lit else 1.0) if _grade_under() else 0.0))
		# Umriss: Figur am Zug in Gold, wählbare Ziele pulsierend
		var ol := Color(0, 0, 0, 0)
		if opt.has("noline"):
			ol = Color(0, 0, 0, 0)
		elif _valid_foes.has(f):
			ol = Color(1.0, 0.35, 0.3, 0.55 + 0.45 * sin(_t * 7.0))
		elif _valid_heroes.has(f):
			ol = Color(0.5, 0.95, 0.4, 0.55 + 0.45 * sin(_t * 7.0))
		elif f == _focus_a and f.side == 0:
			ol = Color(1.0, 0.82, 0.35, 1.0)
		elif f == _focus_a:
			ol = Color(1.0, 0.4, 0.3, 1.0)
		n.mat.set_shader_parameter("outline", ol)
		if opt.has("dbg") and f == _focus_a:
			print("STAGE focus ", f.name, " outline ", ol, " tex ", n.spr.texture.get_size(), " vis ", n.spr.is_visible_in_tree())

static func px_line(ci: CanvasItem, a: Vector2, b: Vector2, col: Color) -> void:
	var x0 := int(a.x)
	var y0 := int(a.y)
	var x1 := int(b.x)
	var y1 := int(b.y)
	var dx := absi(x1 - x0)
	var dy := -absi(y1 - y0)
	var sx := 1 if x0 < x1 else -1
	var sy := 1 if y0 < y1 else -1
	var err := dx + dy
	var guard := 0
	while guard < 800:
		ci.draw_rect(Rect2(x0, y0, 1, 1), col)
		if x0 == x1 and y0 == y1:
			break
		var e2 := 2 * err
		if e2 >= dy:
			err += dy
			x0 += sx
		if e2 <= dx:
			err += dx
			y0 += sy
		guard += 1

func _draw_glow(ci: CanvasItem) -> void:
	var lamp: Color = th["lamp"]
	var tex: Texture2D = _glow_step if opt.has("stepglow") else _glow_soft
	var lamps := _lamp_positions()
	for i in lamps.size():
		var p: Vector2 = lamps[i]
		var fl := 0.8 + 0.2 * sin(_t * 9.0 + float(i) * 2.0) * sin(_t * 5.3 + float(i))
		var r := 44.0 * (0.9 + 0.1 * fl)
		ci.draw_texture_rect(tex, Rect2(p - Vector2(r, r), Vector2(r, r) * 2.0), false, Color(lamp.r, lamp.g, lamp.b, 0.55 * fl))
		ci.draw_texture_rect(tex, Rect2(p - Vector2(10, 10), Vector2(20, 20)), false, Color(1.0, 1.0, 0.9, 0.9 * fl))
		ci.draw_rect(Rect2(p.x - 2, p.y - 3, 5, 7), Color(0.9, 0.9, 0.7, 1.0))
		for k in 3:
			var age := fposmod(_t * 0.6 + float(k) * 0.33 + float(i) * 0.17, 1.0)
			ci.draw_rect(Rect2(roundf(p.x + sin(age * 9.0 + float(k)) * 6.0), roundf(p.y - age * 26.0), 1, 1), Color(1.0, 0.8, 0.4, (1.0 - age) * 0.9))
	# Lichtpfütze unter dem Fokus
	if _focus_a != null:
		var fx := _fx_x(_focus_a)
		var tint := Color(1.0, 0.85, 0.5, 0.35) if _focus_a.side == 0 else Color(1.0, 0.4, 0.3, 0.3)
		ci.draw_texture_rect(tex, Rect2(Vector2(fx - 38, FEET - 14), Vector2(76, 28)), false, tint)
	# Treffer: Halo am Einschlag (aus den Effekten des Hud)
	for it in hud._fx.items:
		var age2 := float(it["age"])
		if age2 < 0.0:
			continue
		var u := clampf(age2 / float(it["dur"]), 0.0, 1.0)
		match String(it["k"]):
			"slash", "bash", "bite", "spark", "muzzle":
				var hp := Vector2(float(it["x"]), float(it["y"]))
				var r2 := 30.0 + 16.0 * u
				ci.draw_texture_rect(tex, Rect2(hp - Vector2(r2, r2), Vector2(r2, r2) * 2.0), false, Color(1.0, 0.8, 0.5, 0.7 * (1.0 - u)))
	# Blitz: ganzflächig additiv (Waves.flash setzt der RainLayer der Vordergrundschicht)
	if Waves.flash > 0.01:
		ci.draw_rect(Rect2(-60, -20, 600, STAGE_H + 40), Color(0.22, 0.26, 0.4, 0.5 * Waves.flash))
