shader_type canvas_item;

// Figuren der Kampfbuehne (Prototyp): ein Material fuer alle, die Werte setzt jede Figur selbst
// (uniform je Figur-Material). Reihenfolge: Entsaettigen, Farbstich, Abdunkeln,
// dann weisser Treffer-Blitz; ausserhalb der Figur ein 1-Pixel-Umriss (Figur im Fokus, waehlbares Ziel).
// Der Umriss braucht transparenten Rand um die Figur: die Texturen sind Ausschnitte mit 2 Pixel Rand.

uniform float alpha : hint_range(0.0, 1.0) = 1.0;  // statt modulate (COLOR enthaelt im fragment schon die Textur, nie "* COLOR")
uniform float flash : hint_range(0.0, 1.0) = 0.0;
uniform float dim : hint_range(0.0, 1.0) = 0.0;
uniform float desat : hint_range(0.0, 1.0) = 0.0;
uniform vec4 tint : source_color = vec4(1.0, 1.0, 1.0, 0.0);
uniform vec4 outline : source_color = vec4(1.0, 0.8, 0.3, 0.0);
uniform vec4 flash_col : source_color = vec4(1.0, 1.0, 1.0, 1.0);

void fragment() {
	vec4 c = texture(TEXTURE, UV);
	vec2 px = TEXTURE_PIXEL_SIZE;
	if (c.a < 0.5) {
		float edge = 0.0;
		if (outline.a > 0.0) {
			edge = max(max(texture(TEXTURE, UV + vec2(px.x, 0.0)).a, texture(TEXTURE, UV - vec2(px.x, 0.0)).a),
				max(texture(TEXTURE, UV + vec2(0.0, px.y)).a, texture(TEXTURE, UV - vec2(0.0, px.y)).a));
		}
		if (edge > 0.5) {
			COLOR = vec4(outline.rgb, outline.a * alpha);
		} else {
			COLOR = vec4(0.0);
		}
	} else {
		vec3 rgb = c.rgb;
		float lum = dot(rgb, vec3(0.299, 0.587, 0.114));
		rgb = mix(rgb, vec3(lum), desat);
		rgb = mix(rgb, rgb * tint.rgb * 1.7, tint.a);
		rgb *= (1.0 - dim);
		rgb = mix(rgb, flash_col.rgb, flash);
		COLOR = vec4(rgb, c.a * alpha);
	}
}
